--- А. Сессии (1.10.0)
docker compose exec -T db sh -c 'psql -U "$POSTGRES_USER" -d ibportal -c "
SELECT count(*) FILTER (WHERE revoked_at IS NULL AND expires_at > now()) AS открытых_сессий,
       count(*) AS строк_в_sessions
FROM sessions;
SELECT count(DISTINCT u.id) AS администраторов_12_часов FROM users u
  JOIN user_roles ur ON ur.user_id = u.id JOIN roles r ON r.id = ur.role_id
  LEFT JOIN role_permissions rp ON rp.role_id = r.id AND rp.permission_code = \$\$users.manage\$\$
  WHERE u.is_active AND (rp.role_id IS NOT NULL OR r.code = \$\$portal_admin\$\$);"'

--- Б. Документы (1.11.0)
docker compose exec -T db sh -c 'psql -U "$POSTGRES_USER" -d ibportal -c "
SELECT v.stage AS этап,
       CASE WHEN v.stage = 14 AND p.cnt > 0 AND p.pending = 0 THEN \$\$все лица согласовали\$\$ WHEN v.stage = 14 THEN \$\$лица не все\$\$ ELSE \$\$\$\$ END AS уточнение,
       count(*) FILTER (WHERE d.deleted_at IS NULL AND d.archived_at IS NULL) AS активных,
       count(*) FILTER (WHERE d.deleted_at IS NULL AND d.archived_at IS NOT NULL) AS в_архиве,
       count(*) FILTER (WHERE d.deleted_at IS NOT NULL) AS удалённых
FROM documents d JOIN document_versions v ON v.id = d.working_version_id AND v.cancelled_at IS NULL
LEFT JOIN LATERAL (SELECT count(*) cnt, count(*) FILTER (WHERE approved_at IS NULL) pending
                   FROM document_parties WHERE version_id = v.id AND removed_at IS NULL) p ON true
GROUP BY 1, 2 ORDER BY 1, 2;
SELECT count(*) FILTER (WHERE approved_version_id IS NOT NULL) AS утверждённых,
       count(*) FILTER (WHERE approved_version_id IS NOT NULL AND working_version_id IS NOT NULL) AS на_пересмотре
FROM documents WHERE deleted_at IS NULL;
SELECT count(*) AS открытых_этапных, count(*) FILTER (WHERE document_state LIKE \$\$%:13\$\$) AS начальника_ИБ_на_13
FROM tasks WHERE document_reason = \$\$stage\$\$ AND completed_at IS NULL AND archived_at IS NULL AND deleted_at IS NULL;
SELECT count(*) AS категория_уже_есть FROM catalog_items
WHERE catalog_code = \$\$task_categories\$\$ AND deleted_at IS NULL AND lower(value) = \$\$контроль по документу\$\$;"'

--- В. Учебные письма (1.12.0)
docker compose exec -T db sh -c 'psql -U "$POSTGRES_USER" -d ibportal -c "
SELECT (SELECT count(*) FROM awareness_letters) AS писем,
       (SELECT count(*) FROM awareness_letters WHERE frozen) AS отправленных,
       (SELECT count(*) FROM awareness_templates) AS шаблонов,
       (SELECT count(*) FROM awareness_letter_events WHERE action = \$\$sent\$\$) AS отметок_отправлено;
SELECT r.name AS реестр, count(rr.id) FILTER (WHERE rr.deleted_at IS NULL) AS записей
FROM registers r LEFT JOIN register_records rr ON rr.register_id = r.id
WHERE r.system_key = \$\$awareness_mailings\$\$ GROUP BY r.name;
SELECT f.label AS поле_уже_есть FROM register_fields f JOIN registers r ON r.id = f.register_id
WHERE r.system_key = \$\$awareness_mailings\$\$ AND f.deleted_at IS NULL AND f.system_key IS NULL
  AND lower(f.label) IN (\$\$шаблон\$\$, \$\$языки\$\$, \$\$получатели\$\$, \$\$файл письма\$\$);"'

   Записать: открытых_сессий, строк_в_sessions, администраторов_12_часов; по этапам 11–16 — активных / в_архиве / удалённых

2. РЕЗЕРВНАЯ КОПИЯ — ОБЯЗАТЕЛЬНО, И КОПИЯ ЗА ПРЕДЕЛЫ ТОМА. БЕЗ НЕЁ migrate НЕ ЗАПУСКАТЬ.
   ./scripts/backup.sh          → последняя строка «Копия создана: <N>»; иначе — стоп.
   B=$(docker volume inspect -f '{{.Mountpoint}}' ibportal_backups)
   N=<имя из строки «Копия создана»>
   sudo tar -C "$B" -cf /tmp/ibportal-backup-$N.tar "$N"
   sha256sum /tmp/ibportal-backup-$N.tar > /tmp/ibportal-backup-$N.tar.sha256
   Перенести оба файла в место хранения вне сервера (по регламенту банка) и там проверить:
     sha256sum -c ibportal-backup-$N.tar.sha256                           → OK
     tar -xf ibportal-backup-$N.tar && (cd $N && sha256sum -c SHA256SUMS) → все OK
   Только после двух OK вне сервера — шаг 3. Имя копии и место хранения — в журнал выкладок.
